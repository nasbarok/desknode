  # -*- coding: utf-8 -*-
  """Derive les DEUX populations de docs/bom.md. Aucune enumeration codee."""
  import re, sys
  src = open("docs/bom.md", encoding="utf-8").read()
  lignes = src.split("\n")
  
  def cells(l):
      l = l.strip()
      if not l.startswith("|"):
          return None
      return [c.strip() for c in l.strip("|").split("|")]
  
  RE_URL = re.compile(r"https?://([A-Za-z0-9.-]+)")
  
  # --- decoupe en tables : entete + separateur + corps
  tables = []
  i = 0
  while i < len(lignes) - 1:
      h = cells(lignes[i]); s = cells(lignes[i+1])
      if h and s and all(re.fullmatch(r":?-{3,}:?", c) for c in s) and len(h) == len(s):
          corps = []
          j = i + 2
          while j < len(lignes):
              c = cells(lignes[j])
              if not c or len(c) != len(h):
                  break
              corps.append((j+1, c)); j += 1
          tables.append((i+1, h, corps)); i = j
      else:
          i += 1
  
  pop1, pop2 = {}, {}
  for lno, h, corps in tables:
      cols = {n.lower(): k for k, n in enumerate(h)}
      isrc = next((k for n, k in cols.items() if n.startswith("source")), None)
      itent = next((k for n, k in cols.items() if "adresse" in n), None)
      for rl, c in corps:
          if isrc is not None:
              for d in RE_URL.findall(c[isrc]):
                  pop1.setdefault(d, []).append(rl)
          if itent is not None:
              for d in RE_URL.findall(c[itent]):
                  pop2.setdefault(d, []).append(rl)
  
  # --- la prose : les domaines HORS table
  en_table = set()
  for lno, h, corps in tables:
      en_table.add(lno); en_table.add(lno+1)
      for rl, _ in corps:
          en_table.add(rl)
  prose = {}
  for k, l in enumerate(lignes, 1):
      if k in en_table:
          continue
      for d in RE_URL.findall(l):
          prose.setdefault(d, []).append(k)
  
  def norm(d):
      return d[4:] if d.startswith("www.") else d
  
  print("TABLES MARKDOWN RECONNUES : %d" % len(tables))
  for lno, h, corps in tables:
      print("  l.%-4d %2d ligne(s)  | %s |" % (lno, len(corps), " | ".join(h)))
  print()
  print("POPULATION 1 — DOMAINES DES CELLULES `Source` DES TABLES DE BOM")
  for d in sorted(pop1):
      print("  %-24s l.%s" % (norm(d), ",".join(str(x) for x in pop1[d])))
  print("  ⇒ %d domaine(s) distinct(s) apres normalisation : %s"
        % (len({norm(d) for d in pop1}), " · ".join(sorted({norm(d) for d in pop1}))))
  print()
  print("POPULATION 1bis — DOMAINES NOMMES EN PROSE, HORS TABLE")
  for d in sorted(prose):
      print("  %-24s l.%s" % (norm(d), ",".join(str(x) for x in prose[d])))
  print()
  print("POPULATION 2 — DOMAINES DE LA TABLE DES ADRESSES TENTEES")
  for d in sorted(pop2):
      print("  %-24s l.%s" % (norm(d), ",".join(str(x) for x in pop2[d])))
  print("  ⇒ %d domaine(s) distinct(s) : %s"
        % (len({norm(d) for d in pop2}), " · ".join(sorted({norm(d) for d in pop2}))))
  print()
  n1 = {norm(d) for d in pop1}; n1b = {norm(d) for d in prose}; n2 = {norm(d) for d in pop2}
  print("INTERSECTION pop1 ∩ pop2 : %s" % (" · ".join(sorted(n1 & n2)) or "(vide)"))
  print("PROSE hors pop1          : %s" % (" · ".join(sorted(n1b - n1)) or "(vide)"))
  print("UNION des trois          : %d domaine(s)" % len(n1 | n1b | n2))
