  | def exif(data):
  |     if data[:2] != b"\xff\xd8":
  |         return None
  |     out = {}
  |     i = 2
  |     while i + 4 <= len(data):
  |         if data[i] != 0xFF:
  |             break
  |         marq = data[i + 1]
  |         if marq in (0xD9, 0xDA):
  |             break
  |         lg = struct.unpack(">H", data[i + 2:i + 4])[0]
  |         seg = data[i + 4:i + 2 + lg]
  |         if marq == 0xE1 and seg[:6] == b"Exif\x00\x00" and "tiff" not in out:
  |             out["tiff"] = True
  |             out.update(tiff(seg[6:]))
  |         if marq == 0xE1 and seg[:len(XMP)] == XMP:
  |             out["xmp"] = True
  |         i += 2 + lg
  |     return out

  | def tiff(t):
  |     try:
  |         e = "<" if t[:2] == b"II" else ">"
  |
  |         def entrees(off):
  |             n = struct.unpack(e + "H", t[off:off + 2])[0]
  |             for k in range(n):
  |                 ent = t[off + 2 + 12 * k: off + 14 + 12 * k]
  |                 tag, typ, cnt = struct.unpack(e + "HHI", ent[:8])
  |                 yield tag, typ, cnt, ent[8:12]
  |
  |         def suivant(off):
  |             n = struct.unpack(e + "H", t[off:off + 2])[0]
  |             return struct.unpack(e + "I", t[off + 2 + 12 * n: off + 6 + 12 * n])[0]
  |
  |         def octets(typ, cnt, val):
  |             lg = TAILLES_TYPE.get(typ, 1) * cnt
  |             if lg <= 4:
  |                 return val[:lg]
  |             o = struct.unpack(e + "I", val)[0]
  |             return t[o:o + lg]
  |
  |         def texte(typ, cnt, val):
  |             if typ != 2:
  |                 return None
  |             o = struct.unpack(e + "I", val)[0] if cnt > 4 else None
  |             raw = t[o:o + cnt] if o is not None else val[:cnt]
  |             return raw.rstrip(b"\x00").decode("latin-1").strip()
  |
  |         out = {}
  |         exif_off = None
  |         off0 = struct.unpack(e + "I", t[4:8])[0]
  |         for tag, typ, cnt, val in entrees(off0):
  |             if tag in IFD0 and texte(typ, cnt, val):
  |                 out[IFD0[tag]] = True
  |             if tag == 0x8825:
  |                 out["gps"] = True
  |             if tag == 0x8769:
  |                 exif_off = struct.unpack(e + "I", val)[0]
  |         if exif_off:
  |             for tag, typ, cnt, val in entrees(exif_off):
  |                 if tag in IFD_EXIF and texte(typ, cnt, val):
  |                     out[IFD_EXIF[tag]] = True
  |                 if tag in PRESENCE_EXIF and cnt > 0:
  |                     out[PRESENCE_EXIF[tag]] = True
  |                 if tag == 0x9286 and octets(typ, cnt, val)[8:].strip(b"\x00 "):
  |                     out["user_comment_non_vide"] = True
  |         off1 = suivant(off0)
  |         if off1:
  |             vues = {tag: (typ, cnt, val) for tag, typ, cnt, val in entrees(off1)}
  |             lg = vues.get(0x0202)
  |             if (0x0201 in vues and lg and struct.unpack(e + ("H" if lg[0] == 3 else "I"), lg[2][:2 if lg[0] == 3 else 4])[0] > 0) \
  |                     or 0x0111 in vues:
  |                 out["vignette_ifd1"] = True
  |         return out
  |     except (struct.error, IndexError):
  |         return {"illisible": True}
