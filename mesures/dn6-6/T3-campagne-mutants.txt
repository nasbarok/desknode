  | #!/usr/bin/env python3
  | # dn6-6 — campagne de mutants d'UNE gate, un mutant par tir : rc, BILAN, [KO ] vus, et pour les gates
  | # a identifiants `(cN)` l'ATTRIBUTION REELLE lue A COLONNE FIXE [9:66] confrontee a CIBLES (import AST).
  | # Usage : campagne.py <racine-du-depot> <gate relative> [--json sortie]
  | import ast, json, re, subprocess, sys, os
  | rac, gate = sys.argv[1], sys.argv[2]
  | out_json = sys.argv[4] if len(sys.argv) > 4 and sys.argv[3] == "--json" else None
  | g = os.path.join(rac, gate)
  | cibles = {}
  | try:
  |     arbre = ast.parse(open(g, encoding="utf-8").read())
  |     for n in ast.walk(arbre):
  |         if (isinstance(n, ast.Assign) and len(n.targets) == 1 and isinstance(n.targets[0], ast.Subscript)
  |                 and isinstance(n.targets[0].value, ast.Name) and n.targets[0].value.id == "CIBLES"):
  |             k = ast.literal_eval(n.targets[0].slice)
  |             cibles[k] = tuple(ast.literal_eval(n.value))
  | except Exception as e:
  |     print("AST:", e)
  | lst = subprocess.run([sys.executable, g, "--liste-mutants"], cwd=rac, capture_output=True, text=True).stdout
  | nums = [int(m.group(1)) for m in re.finditer(r"^\s*(\d+)\s+\S", lst, re.M)]
  | res = []
  | for n in nums:
  |     r = subprocess.run([sys.executable, g, "--mutant", str(n)], cwd=rac, capture_output=True, text=True)
  |     o = r.stdout + r.stderr
  |     b = re.findall(r"^BILAN\s*:.*$", o, re.M)
  |     kos = [l for l in o.split("\n") if re.match(r"^\s*\[KO \]", l)]
  |     ids = set()
  |     for l in kos:
  |         # la colonne FIXE de ctrl() : `  [KO ] ` puis le libelle sur 58 colonnes
  |         mc = re.match(r"^\s*\[KO \] (.{1,58})", l)
  |         lib = mc.group(1) if mc else ""
  |         m = re.match(r"\((\w+)\)", lib)
  |         ids.add(m.group(1) if m else "?")
  |     dec = cibles.get(n)
  |     ecart = None
  |     if dec is not None:
  |         ecart = sorted(set(dec) ^ ids)
  |     res.append({"n": n, "rc": r.returncode, "bilan": b[-1] if b else None, "nko": len(kos),
  |                 "vus": sorted(ids), "cibles": list(dec) if dec is not None else None, "ecart": ecart,
  |                 "ko": [l.strip()[:120] for l in kos]})
  |     print("%3d rc=%s %-22s KO=%d vus=%-22s cibles=%-22s %s" % (
  |         n, r.returncode, (b[-1][8:] if b else "⛔ SANS BILAN"), len(kos), ",".join(sorted(ids)),
  |         ",".join(dec) if dec is not None else "—",
  |         ("" if dec is None else ("✅ 0 ecart" if not ecart else "⛔ ECART %s" % ecart))))
  | if out_json:
  |     json.dump(res, open(out_json, "w"), ensure_ascii=False, indent=1)
  | sains = sum(1 for x in res if x["rc"] == 1 and x["bilan"] and x["nko"] > 0)
  | print("MUTANTS : %d · sains (rc=1+BILAN+[KO ]) : %d · rc=3 : %d · ecarts CIBLES : %d"
  |       % (len(res), sains, sum(1 for x in res if x["rc"] == 3),
  |          sum(1 for x in res if x["ecart"])))
