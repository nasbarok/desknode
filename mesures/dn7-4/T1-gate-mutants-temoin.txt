#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Tire CHAQUE mutant de `tools/verif_placement_dn74.py` INDIVIDUELLEMENT, et
confronte le controle qu'il ROUGIT a la cible qu'il DECLARE.

⛔ « N mutants, N vus rougir » prouve `mutant ⇒ rouge`, ⛔ PAS que le rouge
   tombe la ou le mutant le promet. Les libelles sont lus **A COLONNE FIXE**.
"""
import re
import subprocess
import sys

RACINE = "/home/nasbarok/projects/desknode"
GATE = "tools/verif_placement_dn74.py"

sys.path.insert(0, RACINE + "/tools")
import verif_placement_dn74 as g            # noqa: E402

print("=" * 78)
print("dn7-4 — LES MUTANTS DE `%s`, TIRES UN PAR UN" % GATE)
print("=" * 78)

rc, out = subprocess.getstatusoutput(
    "cd %s && python3 %s --liste-mutants" % (RACINE, GATE))
print("--liste-mutants (rc=%d) :" % (rc >> 8 if rc > 255 else rc))
print(out)
print("-" * 78)

mauvais = []
for n in sorted(g.MUTANTS):
    p = subprocess.run(["python3", GATE, "--mutant", str(n)], cwd=RACINE,
                       stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    txt = p.stdout.decode("utf-8", "replace")
    lignes = txt.split("\n")
    # ⚠️ A COLONNE FIXE : `  [KO ] ` fait 8 colonnes, le libelle en fait 58.
    kos = [l[8:66].strip() for l in lignes if l.startswith("  [KO ]")]
    ids = []
    for k in kos:
        m = re.match(r"\((\w+)\)", k)
        if m:
            ids.append(m.group(1))
    bilan = [l for l in lignes if l.startswith("BILAN")]
    declares = list(g.CIBLES.get(n, ()))
    sain = (p.returncode == 1 and bilan and kos)
    couvre = all(d in ids for d in declares)
    if not (sain and couvre):
        mauvais.append(n)
    print("  %2d  rc=%d  KO=%-2d  %-14s  declare=%-10s  rougit=%s%s"
          % (n, p.returncode, len(kos), (bilan[0] if bilan else "⛔ SANS BILAN"),
             ",".join(declares), ",".join(ids) or "—",
             "" if (sain and couvre) else "   ⛔ ECART"))

p = subprocess.run(["python3", GATE, "--mutant", "999"], cwd=RACINE,
                   stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
print("-" * 78)
print("mutant INCONNU (999) : rc=%d (attendu 2)" % p.returncode)
print("-" * 78)
print("TIR : %d mutant(s), %d ecart(s)" % (len(g.MUTANTS), len(mauvais)))
sys.exit(1 if (mauvais or p.returncode != 2) else 0)
