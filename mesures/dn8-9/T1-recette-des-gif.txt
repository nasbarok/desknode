  (b) `brouillon_gif.py` (le lecteur de structure du § 6) :
  | #!/usr/bin/env python3
  | # brouillon dn8-9 — lecteur de blocs GIF (sera recopie dans tools/verif_vitrine_dn84.py)
  | import struct
  | import sys
  |
  | PIXELS_MAX = 1600
  | MAGIE_GIF = (b"GIF87a", b"GIF89a")
  | APPLI_GIF_PERMISE = b"NETSCAPE2.0"
  |
  |
  | def structure_gif(brut):
  |     if brut[:6] not in MAGIE_GIF or len(brut) < 13:
  |         return None
  |     w, h, packe = struct.unpack("<HHB", brut[6:11])
  |     i = 13 + (3 * 2 ** ((packe & 7) + 1) if packe & 0x80 else 0)
  |     info = {"version": brut[:6].decode("ascii"), "ecran": (w, h),
  |             "images": [], "controles": 0, "netscape": 0, "intrus": [],
  |             "debut_blocs": i, "fin": None}
  |
  |     def sous_blocs(j):
  |         blocs = []
  |         while j < len(brut):
  |             n = brut[j]
  |             j += 1
  |             if n == 0:
  |                 return j, blocs
  |             if j + n > len(brut):
  |                 return None, blocs
  |             blocs.append(brut[j:j + n])
  |             j += n
  |         return None, blocs
  |
  |     while True:
  |         if i >= len(brut):
  |             return None
  |         b = brut[i]
  |         if b == 0x3B:
  |             info["fin"] = i
  |             break
  |         if b == 0x2C:
  |             if i + 11 > len(brut):
  |                 return None
  |             _x, _y, iw, ih, p = struct.unpack("<HHHHB", brut[i + 1:i + 10])
  |             i += 10 + (3 * 2 ** ((p & 7) + 1) if p & 0x80 else 0)
  |             i, _ = sous_blocs(i + 1)
  |             if i is None:
  |                 return None
  |             info["images"].append((iw, ih))
  |         elif b == 0x21:
  |             if i + 2 > len(brut):
  |                 return None
  |             etiquette = brut[i + 1]
  |             i, blocs = sous_blocs(i + 2)
  |             if i is None:
  |                 return None
  |             if etiquette == 0xF9:
  |                 info["controles"] += 1
  |             elif etiquette == 0xFF and blocs and blocs[0] == APPLI_GIF_PERMISE:
  |                 info["netscape"] += 1
  |             elif etiquette == 0xFF:
  |                 info["intrus"].append("application %r" % (blocs[0] if blocs else b"",))
  |             elif etiquette == 0xFE:
  |                 info["intrus"].append("commentaire 0xFE")
  |             elif etiquette == 0x01:
  |                 info["intrus"].append("texte brut 0x01")
  |             else:
  |                 info["intrus"].append("inconnue 0x%02X" % etiquette)
  |         else:
  |             return None
  |     if not info["images"]:
  |         return None
  |     return info
  |
  |
  | if __name__ == "__main__":
  |     for p in sys.argv[1:]:
  |         b = open(p, "rb").read()
  |         g = structure_gif(b)
  |         if g is None:
  |             print(p, "ILLISIBLE")
  |             continue
  |         print(p, len(b), g["version"], g["ecran"], "images", len(g["images"]),
  |               "tailles", sorted(set(g["images"])), "F9", g["controles"],
  |               "NETSCAPE", g["netscape"], "intrus", g["intrus"],
  |               "fin", g["fin"], "apres_trailer", len(b) - g["fin"] - 1)
