  | URL=$(python3 - "$DN/README.md" <<'PY'
  | import re, sys
  | txt = open(sys.argv[1], encoding="utf-8").read()
  | m = re.search(r"(?m)^## +Demo\s*$", txt)
  | if not m:
  |     print("SECTION-ABSENTE"); sys.exit(0)
  | suite = re.search(r"(?m)^## ", txt[m.end():])
  | sec = txt[m.end():m.end() + suite.start()] if suite else txt[m.end():]
  | liens = re.findall(r"(?<!!)\[[^\]]*\]\((https?://[^)\s]+)\)", sec)
  | print(" ".join(liens) if liens else "AUCUN-LIEN")
  | PY
  | )

  | echo "  attendue (mesuree le 2026-09-13) : $ATTENDUE"

  | code watch-temoin "https://www.youtube.com/watch?v=$TEMOIN"
  | echo
  | echo "── (3) CE QUE RENDENT LES CORPS ──"
  | echo "  oembed (vrai lien), champs :"

  | echo "  page publique (temoin) :"

  | done
  | echo "  taille des corps : $(stat -c '%n=%s' "$O"/*.corps | sed "s#$O/##g" | tr '\n' ' ')"
  | echo "=== fin $(date -Iseconds) ==="
