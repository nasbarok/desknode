# -*- coding: utf-8 -*-
"""Comptage a BORNES DE MOT — instrument de dn5-1 (mesure capturee, ⛔ pas un outil durable).

⛔ NE PAS grep -c : « tier » est une SOUS-CHAINE de « entier », « boitier »,
   « chantier », « tiers ». Le compte sous-chaine est FAUX et on le MESURE ici.
Chaque chiffre est publie AVEC son motif et AVEC ses occurrences en contexte.
"""
import re, subprocess, sys, collections

MOTIFS = {
    r"\bpaliers?\b":  "palier / paliers  (mot entier, insensible a la casse)",
    r"\btiers?\b":    "tier / tiers      (mot entier, insensible a la casse) — ⚠️ capture AUSSI le francais « tiers »",
}
SOUS_CHAINE = "tier"

def lire(ref, chemin):
    r = subprocess.run(["git", "show", f"{ref}:{chemin}"],
                       capture_output=True)
    if r.returncode != 0:
        return None
    return r.stdout.decode("utf-8", errors="replace")

def contexte(txt, m, large=42):
    a = max(0, m.start() - large); b = min(len(txt), m.end() + large)
    return txt[a:b].replace("\n", " ⏎ ")

def rapport(ref, chemin):
    txt = lire(ref, chemin)
    print(f"\n── {ref}:{chemin}")
    if txt is None:
        print("   ⛔ ABSENT de cette ref")
        return
    print(f"   taille : {len(txt.encode('utf-8'))} o · {len(txt.splitlines())} lignes")
    for motif, libelle in MOTIFS.items():
        occ = list(re.finditer(motif, txt, re.IGNORECASE))
        print(f"   motif {motif!r:18} = {len(occ):3d}   [{libelle}]")
        formes = collections.Counter(m.group(0) for m in occ)
        if formes:
            print(f"        formes vues : {dict(formes)}")
        for m in occ[:12]:
            print(f"        · …{contexte(txt, m)}…")
        if len(occ) > 12:
            print(f"        … ({len(occ)-12} autre(s) non imprimee(s))")
    # LE PIEGE, MESURE : le meme mot compte en SOUS-CHAINE
    occ = list(re.finditer(SOUS_CHAINE, txt, re.IGNORECASE))
    formes = collections.Counter()
    for m in occ:
        a = m.start(); b = m.end()
        while a > 0 and (txt[a-1].isalpha() or txt[a-1] in "àâäçéèêëîïôöùûüÿœæ"): a -= 1
        while b < len(txt) and (txt[b].isalpha() or txt[b] in "àâäçéèêëîïôöùûüÿœæ"): b += 1
        formes[txt[a:b]] += 1
    print(f"   ⚠️ PIEGE — sous-chaine {SOUS_CHAINE!r} (grep -oi) = {len(occ):3d}   ⛔ CE CHIFFRE EST FAUX")
    print(f"        mots reellement porteurs : {dict(formes)}")

if __name__ == "__main__":
    print("=" * 78)
    print("COMPTAGE A BORNES DE MOT — dn5-1 / T0 (rejeu du cadrage §2 §4 §5)")
    print("=" * 78)

        for chemin in sys.argv[2].split(","):
            rapport(ref, chemin)
